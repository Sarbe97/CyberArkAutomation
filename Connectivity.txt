# --- Configuration ---
$InputFile = ".\servers.txt"
$OutputFile = ".\CPM_Matrix_Report.csv"

# Define the ports we care about
$PortsToTest = @{
    "SMB_445"    = 445
    "RPC_135"    = 135
    "LDAP_389"   = 389
    "WinRM_5986" = 5986
}

if (-not (Test-Path $InputFile)) {
    Write-Host "Error: $InputFile not found!" -ForegroundColor Red
    return
}

$Servers = Get-Content -Path $InputFile
$FinalReport = @()

foreach ($Server in $Servers) {
    $Server = $Server.Trim()
    if ([string]::IsNullOrWhiteSpace($Server)) { continue }
    
    Write-Host "Processing $Server..." -ForegroundColor Cyan
    
    # Create a custom object for this specific server
    $ServerObj = [PSCustomObject]@{
        ServerName = $Server
        Ping Status = "Unknown"
    }

    # 1. Quick Ping Test
    if (Test-Connection -ComputerName $Server -Count 1 -Quiet) {
        $ServerObj.'Ping Status' = "Online"
    } else {
        $ServerObj.'Ping Status' = "Offline/No Reply"
    }

    # 2. Test each Port and add as a column
    foreach ($Key in $PortsToTest.Keys) {
        $Port = $PortsToTest[$Key]
        $Check = Test-NetConnection -ComputerName $Server -Port $Port -InformationLevel Quiet -WarningAction SilentlyContinue
        $ServerObj | Add-Member -MemberType NoteProperty -Name $Key -Value (if ($Check) { "OPEN" } else { "BLOCKED" })
    }

    # 3. WMI/RPC Dynamic Range Check
    try {
        $Wmi = Get-WmiObject -ComputerName $Server -Class Win32_OperatingSystem -ErrorAction Stop -ConnectTimeout 3
        $ServerObj | Add-Member -MemberType NoteProperty -Name "WMI_Dynamic_RPC" -Value "SUCCESS"
    } catch {
        $ServerObj | Add-Member -MemberType NoteProperty -Name "WMI_Dynamic_RPC" -Value "FAILED"
    }

    $FinalReport += $ServerObj
}

# Export to CSV and Display
$FinalReport | Export-Csv -Path $OutputFile -NoTypeInformation
$FinalReport | Format-Table -AutoSize
Write-Host "`nReport generated: $OutputFile" -ForegroundColor Green
