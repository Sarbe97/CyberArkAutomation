# ============================================================
# 3. Delete Discovered Accounts (Bulk from CSV)
# ============================================================
function Remove-CACDiscoveredAccounts {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory=$false, HelpMessage="Path to the input CSV file")]
        [string]$CsvPath
    )

    Write-Log "Started Remove-CACDiscoveredAccounts()" "DEBUG"

    try {
        Write-Host ""
        Write-Host "===== Bulk Delete Discovered Accounts =====" -ForegroundColor Cyan
        Write-Host ""
        
        # 1. Handle Input CSV
        if ([string]::IsNullOrWhiteSpace($CsvPath)) {
            $CsvPath = Read-Host "Enter path to CSV file (must contain 'Id' column)"
        }

        # Remove quotes if user copied path as "C:\Path"
        $CsvPath = $CsvPath -replace '"', ''

        if (-not (Test-Path $CsvPath)) {
            Write-Host "Error: File not found at $CsvPath" -ForegroundColor Red
            return
        }

        $csvData = Import-Csv -Path $CsvPath
        
        if ($csvData.Count -eq 0) {
            Write-Host "Error: CSV file is empty." -ForegroundColor Red
            return
        }

        # Validate required column exists
        $firstRow = $csvData | Select-Object -First 1
        if (-not $firstRow.PSObject.Properties['Id']) {
            Write-Host "Error: Input CSV must contain a column named 'Id'." -ForegroundColor Red
            Write-Host "Please use the export from Get-CACDiscoveredAccounts as a template." -ForegroundColor Yellow
            return
        }

        Write-Host "Loaded $($csvData.Count) records to process." -ForegroundColor Cyan
        $confirm = Read-Host "Are you sure you want to attempt deletion for these accounts? (Y/N)"
        if ($confirm -ne 'Y' -and $confirm -ne 'y') {
            Write-Host "Operation cancelled." -ForegroundColor Yellow
            return
        }

        # 2. Process Deletions
        $results = [System.Collections.Generic.List[PSCustomObject]]::new()
        $successCount = 0
        $failCount = 0

        foreach ($row in $csvData) {
            $accountId = $row.Id
            $acctName = if ($row.UserName) { $row.UserName } else { $accountId }
            
            # Clone row to avoid modifying the loop iterator directly and ensure distinct object
            $outputRow = $row | Select-Object * try {
                Write-Host "Deleting account ID: $accountId ($acctName)... " -NoNewline

                # CyberArk API: DELETE /API/DiscoveredAccounts/{id}
                $endpoint = "/API/DiscoveredAccounts/$accountId"
                
                # Invoke-CACAPIRequest usually returns the response or throws on error
                # DELETE returns 204 No Content on success, which Invoke-RestMethod usually returns as null/empty
                Invoke-CACAPIRequest -Method DELETE -Endpoint $endpoint | Out-Null

                Write-Host "Success" -ForegroundColor Green
                
                # Append Status columns
                $outputRow | Add-Member -MemberType NoteProperty -Name "DeletionStatus" -Value "Success" -Force
                $outputRow | Add-Member -MemberType NoteProperty -Name "DeletionMessage" -Value "Deleted successfully" -Force
                $successCount++
            }
            catch {
                $errMsg = $_.Exception.Message
                # Handle specific CyberArk API error structures if encapsulated
                if ($_.Exception.Response) {
                    # Try to read stream if it's a web exception
                    # (This depends on how Invoke-CACAPIRequest handles errors, but this is a safe generic catch)
                }

                Write-Host "Failed - $errMsg" -ForegroundColor Red
                
                $outputRow | Add-Member -MemberType NoteProperty -Name "DeletionStatus" -Value "Failed" -Force
                $outputRow | Add-Member -MemberType NoteProperty -Name "DeletionMessage" -Value $errMsg -Force
                $failCount++
                Write-Log "Failed to delete discovered account $accountId: $errMsg" "ERROR"
            }

            $results.Add($outputRow)
        }

        # 3. Export Results
        Write-Host ""
        Write-Host "===== Deletion Summary =====" -ForegroundColor Cyan
        Write-Host "Total Processed: $($results.Count)"
        Write-Host "Successful:      $successCount" -ForegroundColor Green
        Write-Host "Failed:          $failCount" -ForegroundColor Red
        Write-Host ""

        $outputDir = Get-CACOutputDir
        $timestamp = Get-Date -Format "yyyyMMdd_HHmmss"
        $outputFile = "$outputDir/delete_discovered_results_$timestamp.csv"

        $results | Export-Csv -Path $outputFile -NoTypeInformation -Encoding UTF8
        
        Write-Host "Results exported to: $outputFile" -ForegroundColor Green
        Write-Log "Bulk delete completed. Success: $successCount, Failed: $failCount. Results: $outputFile" "INFO"
    }
    catch {
        Write-Log "Error in Remove-CACDiscoveredAccounts(): $($_.Exception.Message)" "ERROR"
        Write-Host "Critical Error: $($_.Exception.Message)" -ForegroundColor Red
    }
}
