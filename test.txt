# =============================================================================
# Generic Service Accounts vs CyberArk Matching Analysis
# =============================================================================

# -----------------------------------------------------------------------------
# 1. Input / Output Paths - Paste your full file paths here
# -----------------------------------------------------------------------------
$adCsvPath = "C:\Path\To\Filtered_ADServiceAccounts.csv"
$caCsvPath = "C:\Path\To\CyberArkAccounts.csv"

# Optional: Output CSV path. Leave empty ("") to auto-create in the same folder where this script is located
$outCsv    = ""

# -----------------------------------------------------------------------------
# 2. File Verification & Path Resolution
# -----------------------------------------------------------------------------
if (-not (Test-Path $adCsvPath)) {
    Write-Error "AD CSV file not found at: '$adCsvPath'. Please paste a valid path."
    exit 1
}

if (-not (Test-Path $caCsvPath)) {
    Write-Error "CyberArk CSV file not found at: '$caCsvPath'. Please paste a valid path."
    exit 1
}

# Resolve script directory (works when running as script, module, or from PowerShell console)
$scriptDir = $PSScriptRoot
if ([string]::IsNullOrWhiteSpace($scriptDir)) {
    $scriptDir = Split-Path -Parent $MyInvocation.MyCommand.Definition -ErrorAction SilentlyContinue
}
if ([string]::IsNullOrWhiteSpace($scriptDir)) {
    $scriptDir = (Get-Location).Path
}

if ([string]::IsNullOrWhiteSpace($outCsv)) {
    $adBase = [System.IO.Path]::GetFileNameWithoutExtension($adCsvPath)
    $outCsv = Join-Path $scriptDir "${adBase}_MatchBreakdown.csv"
}

Write-Host "Loading AD Service Accounts from: $adCsvPath" -ForegroundColor Cyan
Write-Host "Loading CyberArk Inventory from : $caCsvPath" -ForegroundColor Cyan

$adAccounts = Import-Csv -Path $adCsvPath
$caAccounts = Import-Csv -Path $caCsvPath

# -----------------------------------------------------------------------------
# 3. Build CyberArk Lookup Table (Grouped by lowercase username)
# -----------------------------------------------------------------------------
$caLookup = @{}
foreach ($ca in $caAccounts) {
    if (-not $ca.Username) { continue }
    $u = $ca.Username.ToLower().Trim()
    if (-not $caLookup.ContainsKey($u)) {
        $caLookup[$u] = [System.Collections.Generic.List[object]]::new()
    }
    $caLookup[$u].Add($ca)
}

Write-Host "Indexed $($caAccounts.Count) CyberArk accounts ($($caLookup.Count) unique usernames)." -ForegroundColor Cyan
Write-Host "Evaluating $($adAccounts.Count) AD service accounts..." -ForegroundColor Cyan

# -----------------------------------------------------------------------------
# 4. Cross-Reference AD Accounts against CyberArk
# -----------------------------------------------------------------------------
$results = foreach ($ad in $adAccounts) {
    $uKey  = if ($ad.Username)   { $ad.Username.ToLower().Trim() }   else { "" }
    $fqdn  = if ($ad.DomainFQDN) { $ad.DomainFQDN.ToLower().Trim() } else { "" }
    $short = if ($ad.Domain)     { $ad.Domain.ToLower().Trim() }     else { "" }

    $isUsernameMatch       = $false
    $isUserAndDomainMatch  = $false
    $matchStatus           = "Not_In_CyberArk"
    $caSafe                = ""
    $caPlatform            = ""
    $caAddress             = ""

    if ($uKey -and $caLookup.ContainsKey($uKey)) {
        $isUsernameMatch = $true

        # Check if any CyberArk account matching this username also matches domain/address
        $domainMatches = @($caLookup[$uKey] | Where-Object {
            $addr = if ($_.Address) { ($_.Address).ToLower().Trim() } else { "" }
            ($addr -and $fqdn  -and $addr -eq $fqdn) -or
            ($addr -and $short -and $addr -eq $short) -or
            ($addr -and $fqdn  -and $fqdn.StartsWith($addr)) -or
            ($addr -and $fqdn  -and $addr.EndsWith($fqdn)) -or
            ($addr -and $fqdn  -and $addr.Contains($fqdn)) -or
            ($addr -and $fqdn  -and $fqdn.Contains($addr))
        })

        if ($domainMatches.Count -gt 0) {
            # Case 1: Both Username AND Domain match
            $isUserAndDomainMatch = $true
            $matchStatus          = "Both_Matched"
            $caSafe               = ($domainMatches | ForEach-Object { $_.SafeName }   | Select-Object -Unique) -join "; "
            $caPlatform           = ($domainMatches | ForEach-Object { $_.PlatformId } | Select-Object -Unique) -join "; "
            $caAddress            = ($domainMatches | ForEach-Object { $_.Address }    | Select-Object -Unique) -join "; "
        }
        else {
            # Case 2: Username matches, but address/domain does not match
            $matchStatus = "Username_Only"
            $allForeign  = $caLookup[$uKey]
            $caSafe      = ($allForeign | ForEach-Object { $_.SafeName }   | Select-Object -Unique) -join "; "
            $caPlatform  = ($allForeign | ForEach-Object { $_.PlatformId } | Select-Object -Unique) -join "; "
            $caAddress   = ($allForeign | ForEach-Object { $_.Address }    | Select-Object -Unique) -join "; "
        }
    }

    # Build output object retaining all original AD properties + adding match columns
    $rowProps = [ordered]@{}
    foreach ($prop in $ad.PSObject.Properties) {
        $rowProps[$prop.Name] = $prop.Value
    }

    $rowProps["MatchStatus"]              = $matchStatus
    $rowProps["UsernameMatched"]          = $isUsernameMatch
    $rowProps["UsernameAndDomainMatched"] = $isUserAndDomainMatch
    $rowProps["CyberArkSafe"]             = $caSafe
    $rowProps["CyberArkPlatform"]         = $caPlatform
    $rowProps["CyberArkAddress"]          = $caAddress

    [PSCustomObject]$rowProps
}

# -----------------------------------------------------------------------------
# 5. Export Results & Console Summary
# -----------------------------------------------------------------------------
$results | Export-Csv -Path $outCsv -NoTypeInformation -Encoding UTF8

$cntBoth        = @($results | Where-Object { $_.MatchStatus -eq "Both_Matched" }).Count
$cntUserOnly    = @($results | Where-Object { $_.MatchStatus -eq "Username_Only" }).Count
$cntNotInCA     = @($results | Where-Object { $_.MatchStatus -eq "Not_In_CyberArk" }).Count

Write-Host ""
Write-Host "================ Match Summary ================" -ForegroundColor Cyan
Write-Host "Both Username & Domain Matched : $cntBoth" -ForegroundColor Green
Write-Host "Username Only Matched          : $cntUserOnly" -ForegroundColor Yellow
Write-Host "Not In CyberArk                : $cntNotInCA" -ForegroundColor Red
Write-Host "Total AD Service Accounts      : $($results.Count)"
Write-Host "===============================================" -ForegroundColor Cyan
Write-Host "Output saved to: $outCsv" -ForegroundColor Green
